-- ============================================================
-- AUTO SAVE / LOAD SYSTEM
-- getgenv() persists across re-executes in the same session (like
-- the reference script). writefile/readfile persist across sessions.
-- "Desync (Work in Progress)" is never saved or restored.
-- ============================================================
local NO_SAVE = { ["Desync (Work in Progress)"] = true }
local HttpService = game:GetService("HttpService")
local SAVE_FILE = "ChlipHubSave.json"

-- Init persistent save table via getgenv (survives re-execute)
if type(getgenv().CHLIP_SAVED_STATES) ~= "table" then
    getgenv().CHLIP_SAVED_STATES = {}
end
local savedStates = getgenv().CHLIP_SAVED_STATES

-- Try to load from file on first ever run (empty table = first run)
if not getgenv().CHLIP_FILE_LOADED then
    getgenv().CHLIP_FILE_LOADED = true
    pcall(function()
        local raw = readfile(SAVE_FILE)
        if raw and raw ~= "" then
            local data = HttpService:JSONDecode(raw)
            if type(data) == "table" then
                for k, v in pairs(data) do
                    savedStates[k] = v
                end
            end
        end
    end)
    savedStates["Desync (Work in Progress)"] = nil
end

local function saveState(name, state)
    if NO_SAVE[name] then return end
    savedStates[name] = state
    -- Write to file so it persists across game sessions too
    pcall(function()
        local toSave = {}
        for k, v in pairs(savedStates) do
            if not NO_SAVE[k] then toSave[k] = v end
        end
        writefile(SAVE_FILE, HttpService:JSONEncode(toSave))
    end)
end

local function loadSavedStates()
    for name, state in pairs(savedStates) do
        if NO_SAVE[name] then continue end
        if state == true then
            _G.chlip_toggleStates[name] = true
            if _G.chlip_toggleCallbacks[name] then
                task.spawn(function()
                    pcall(_G.chlip_toggleCallbacks[name], true)
                end)
            end
        end
    end
end

local function antiLag_setupCharacterEvents(character)
	if antiLagConnections[character] then return end
	local connections = {}
	table.insert(connections, character.ChildAdded:Connect(function(child)
		if child:IsA("Accessory") then child:Destroy() end
		if child:IsA("Tool") then
			for _, effect in pairs(child:GetDescendants()) do
				if effect:IsA("ParticleEmitter") or effect:IsA("Trail") or effect:IsA("Beam") or
				   effect:IsA("PointLight") or effect:IsA("SpotLight") or effect:IsA("SurfaceLight") or
				   effect:IsA("Fire") or effect:IsA("Smoke") or effect:IsA("Sparkles") or
				   effect:IsA("Highlight") or effect:IsA("Decal") or effect:IsA("Texture") or
				   effect:IsA("SpecialMesh") then
					effect:Destroy()
				end
			end
			child.DescendantAdded:Connect(function(descendant)
				if descendant.Name:sub(1,3) == "CH_" then return end
				if descendant:IsA("ParticleEmitter") or descendant:IsA("Trail") or descendant:IsA("Beam") or
				   descendant:IsA("PointLight") or descendant:IsA("SpotLight") or descendant:IsA("SurfaceLight") or
				   descendant:IsA("Fire") or descendant:IsA("Smoke") or descendant:IsA("Sparkles") or
				   descendant:IsA("Highlight") or descendant:IsA("Decal") or descendant:IsA("Texture") or
				   descendant:IsA("SpecialMesh") then
					descendant:Destroy()
				end
			end)
		end
	end))
	table.insert(connections, character.DescendantAdded:Connect(function(descendant)
		if descendant.Name:sub(1,3) == "CH_" then return end
		if descendant:IsA("Accessory") or
		   descendant:IsA("Shirt") or descendant:IsA("Pants") or descendant:IsA("ShirtGraphic") or
		   descendant:IsA("CharacterMesh") or descendant:IsA("SpecialMesh") or
		   descendant:IsA("ParticleEmitter") or descendant:IsA("Trail") or descendant:IsA("Beam") or
		   descendant:IsA("PointLight") or descendant:IsA("SpotLight") or descendant:IsA("SurfaceLight") or
		   descendant:IsA("Fire") or descendant:IsA("Smoke") or descendant:IsA("Sparkles") or
		   descendant:IsA("Highlight") then
			descendant:Destroy()
		end
		if descendant:IsA("Decal") or descendant:IsA("Texture") then
			local isFace = descendant.Name:lower():find("face") or (descendant:IsA("Decal") and descendant.Texture:lower():find("face"))
			local parentHead = descendant.Parent and descendant.Parent.Name:lower() == "head"
			if not (isFace and parentHead) then descendant:Destroy() end
		end
	end))
	antiLagConnections[character] = connections
	character.Destroying:Connect(function()
		if antiLagConnections[character] then
			for _, conn in pairs(antiLagConnections[character]) do conn:Disconnect() end
			antiLagConnections[character] = nil
		end
	end)
end

local function antiLag_onCharacterAdded(character)
	antiLag_optimizeCharacter(character)
	antiLag_setupCharacterEvents(character)
end

local function antiLag_disconnectAll()
	for char, conns in pairs(antiLagConnections) do
		for _, conn in pairs(conns) do conn:Disconnect() end
		antiLagConnections[char] = nil
	end
	if antiLagHeartbeat then antiLagHeartbeat:Disconnect(); antiLagHeartbeat = nil end
	if antiLagCharConn then antiLagCharConn:Disconnect(); antiLagCharConn = nil end
end

			if a.plot == plotName then espBest_animals[uid] = nil end
		end
		return
	end
	local channel = espBest_stealthGet(plotName)
	if not channel then
		for uid, a in pairs(espBest_animals) do
			if a.plot == plotName then espBest_animals[uid] = nil end
		end
		return
	end
	local list = espBest_stealthGetFrom(channel, "AnimalList")
	if not list or type(list) ~= "table" then
		for uid, a in pairs(espBest_animals) do
			if a.plot == plotName then espBest_animals[uid] = nil end
		end
		return
	end
	local alive = {}
	local okAnimalsShared, AnimalsShared = pcall(require, RS.Shared:WaitForChild("Animals"))
	local okAnimalsData, AnimalsData = pcall(require, RS.Datas:WaitForChild("Animals"))
	for slot, data in pairs(list) do
		if type(data) ~= "table" then continue end
		local uid = plotName .. "_" .. tostring(slot)
		alive[uid] = true
		local okGen, gen = pcall(function()
			return AnimalsShared:GetGeneration(data.Index, data.Mutation, data.Traits, nil)
		end)
		if not okGen or not okAnimalsShared or not gen then espBest_animals[uid] = nil; continue end
		local displayName = (okAnimalsData and AnimalsData and AnimalsData[data.Index] and (AnimalsData[data.Index].DisplayName or AnimalsData[data.Index].Name)) or tostring(data.Index)
		if espBest_shouldIgnore(displayName) then espBest_animals[uid] = nil; continue end

		else
			espBest_animals[uid] = { uid = uid, name = displayName, gen = gen, plot = plotName, slot = tostring(slot) }
		end
	end
	for uid, a in pairs(espBest_animals) do
		if a.plot == plotName and not alive[uid] then espBest_animals[uid] = nil end
	end

local antiRagdoll_mode = nil
local antiRagdoll_connections = {}
local antiRagdoll_charData = {}
local antiRagdoll_isBoosting = false

local AR_BOOST_SPEED = 400
local AR_DEFAULT_SPEED = 16

local function antiRagdoll_cacheChar()
	local char = player.Character
	if not char then return false end
	local hum = char:FindFirstChildOfClass("Humanoid")
	local root = char:FindFirstChild("HumanoidRootPart")
	if not hum or not root then return false end
	antiRagdoll_charData = { character = char, humanoid = hum, root = root }
	return true
end

local function antiRagdoll_disconnectAll()
	for _, conn in ipairs(antiRagdoll_connections) do
		pcall(function() conn:Disconnect() end)
	end
	antiRagdoll_connections = {}
end

local function antiRagdoll_isRagdolled()
	if not antiRagdoll_charData.humanoid then return false end
	local state = antiRagdoll_charData.humanoid:GetState()
	local ragdollStates = {
		[Enum.HumanoidStateType.Physics] = true,
		[Enum.HumanoidStateType.Ragdoll] = true,
		[Enum.HumanoidStateType.FallingDown] = true,
	}
	if ragdollStates[state] then return true end
	local endTime = player:GetAttribute("RagdollEndTime")
	if endTime and (endTime - workspace:GetServerTimeNow()) > 0 then return true end
	return false
end

local function antiRagdoll_forceExit()
	if not antiRagdoll_charData.humanoid or not antiRagdoll_charData.root then return end
	pcall(function()
		player:SetAttribute("RagdollEndTime", workspace:GetServerTimeNow())
	end)
	for _, d in ipairs(antiRagdoll_charData.character:GetDescendants()) do
		if d:IsA("BallSocketConstraint") or (d:IsA("Attachment") and d.Name:find("RagdollAttachment")) then
			d:Destroy()
		end
	end
	if not antiRagdoll_isBoosting then
		antiRagdoll_isBoosting = true
		antiRagdoll_charData.humanoid.WalkSpeed = AR_BOOST_SPEED
	end
	if antiRagdoll_charData.humanoid.Health > 0 then
		antiRagdoll_charData.humanoid:ChangeState(Enum.HumanoidStateType.Running)
	end
	antiRagdoll_charData.root.Anchored = false
end

local function antiRagdoll_heartbeatLoop()
	while antiRagdoll_mode == "v1" do
		task.wait()
		local ragdolled = antiRagdoll_isRagdolled()
		if ragdolled then
			antiRagdoll_forceExit()
		elseif antiRagdoll_isBoosting and not ragdolled then
			antiRagdoll_isBoosting = false
			if antiRagdoll_charData.humanoid then
				antiRagdoll_charData.humanoid.WalkSpeed = AR_DEFAULT_SPEED
			end
		end
	end
end

local function antiRagdoll_enable()
	if antiRagdoll_mode == "v1" then return end
	if not antiRagdoll_cacheChar() then return end
	antiRagdoll_mode = "v1"
	local camConn = game:GetService("RunService").RenderStepped:Connect(function()
		local cam = workspace.CurrentCamera
		if cam and antiRagdoll_charData.humanoid then
			cam.CameraSubject = antiRagdoll_charData.humanoid
		end
	end)
	table.insert(antiRagdoll_connections, camConn)
	local respawnConn = player.CharacterAdded:Connect(function()
		antiRagdoll_isBoosting = false
		task.wait(0.5)
		antiRagdoll_cacheChar()
	end)
	table.insert(antiRagdoll_connections, respawnConn)
	task.spawn(antiRagdoll_heartbeatLoop)
end

local function antiRagdoll_disable()
	antiRagdoll_mode = nil
	if antiRagdoll_isBoosting and antiRagdoll_charData.humanoid then
		antiRagdoll_charData.humanoid.WalkSpeed = AR_DEFAULT_SPEED
	end
	antiRagdoll_isBoosting = false
	antiRagdoll_disconnectAll()
	antiRagdoll_charData = {}
end

toggleCallbacks["Anti Ragdoll"] = function(isOn)
	if isOn then
		antiRagdoll_enable()
		print("Anti Ragdoll: ON")
	else
		antiRagdoll_disable()
		print("Anti Ragdoll: OFF")
	end
end

			or state == Enum.HumanoidStateType.FallingDown
			or state == Enum.HumanoidStateType.Jumping

			if tick() - ij_lastJumpTime >= IJ_HOLD_DELAY then
				ij_lastJumpTime = tick()
				ij_jumpsUsed += 1
				ij_currentRoot.AssemblyLinearVelocity = Vector3.new(
					ij_currentRoot.AssemblyLinearVelocity.X,

local disableAnim_descConn = nil

local function disableAnim_disableAnimator(animator)
	for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
		track:Stop(0)
	end
	animator.AnimationPlayed:Connect(function(track)
		track:Stop(0)
	end)
end

local function disableAnim_processCharacter(character)
	local ownerPlayer = game:GetService("Players"):GetPlayerFromCharacter(character)
	if ownerPlayer == player then return end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if humanoid then
		local animator = humanoid:FindFirstChildOfClass("Animator")
		if animator then
			disableAnim_disableAnimator(animator)
		else
			humanoid.ChildAdded:Connect(function(child)
				if child:IsA("Animator") then disableAnim_disableAnimator(child) end
			end)
		end
	end

	for _, child in ipairs(character:GetDescendants()) do
		if child:IsA("AnimationController") then
			local animator = child:FindFirstChildOfClass("Animator")
			if animator then disableAnim_disableAnimator(animator) end
		end
	end

	character.DescendantAdded:Connect(function(descendant)
		if descendant:IsA("Animator") then
			local parent = descendant.Parent
			if parent:IsA("AnimationController") or parent:IsA("Humanoid") then
				disableAnim_disableAnimator(descendant)
			end
		end
	end)
end

local function disableAnim_start()
	for _, model in ipairs(workspace:GetDescendants()) do
		if model:IsA("Model") then disableAnim_processCharacter(model) end
	end

	disableAnim_descConn = workspace.DescendantAdded:Connect(function(descendant)
		if descendant:IsA("Model") then
			task.defer(function() disableAnim_processCharacter(descendant) end)
		end
	end)

	local Players = game:GetService("Players")
	disableAnim_playerAddedConn = Players.PlayerAdded:Connect(function(plr)
		if plr == player then return end
		plr.CharacterAdded:Connect(function(character)
			task.defer(function() disableAnim_processCharacter(character) end)
		end)
		if plr.Character then disableAnim_processCharacter(plr.Character) end
	end)

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= player and plr.Character then
			disableAnim_processCharacter(plr.Character)
		end
	end
end

local function disableAnim_stop()
	if disableAnim_descConn then disableAnim_descConn:Disconnect(); disableAnim_descConn = nil end
	if disableAnim_playerAddedConn then disableAnim_playerAddedConn:Disconnect(); disableAnim_playerAddedConn = nil end
end

toggleCallbacks["Disable Brainrot Animation"] = function(isOn)

		disableAnim_start()

		print("Disable Brainrot Animation: OFF")

local function antiBee_nuke(obj)
	if obj and obj.Parent and ANTI_BEE_BAD_NAMES[obj.Name] then

	end
end

local function antiBee_disconnectAll()
	for _, conn in ipairs(antiBee_connections) do
		if typeof(conn) == "RBXScriptConnection" then conn:Disconnect() end
	end
	table.clear(antiBee_connections)
end

local function antiBee_startFOV()
	if antiBee_fovConn then return end
	antiBee_fovConn = game:GetService("RunService").RenderStepped:Connect(function()
		local cam = workspace.CurrentCamera
		if cam and cam.FieldOfView ~= 70 then cam.FieldOfView = 70 end
	end)

	if antiBee_fovConn then antiBee_fovConn:Disconnect(); antiBee_fovConn = nil end
end

local function antiBee_protectControls()
	if antiBee_controlsProtected then return end
	pcall(function()
		local ps = player:WaitForChild("PlayerScripts")
		local pm = ps:FindFirstChild("PlayerModule")
		if not pm then return end
		local Controls = require(pm):GetControls()
		if not Controls then return end
		if not antiBee_originalMoveFunction then
			antiBee_originalMoveFunction = Controls.moveFunction
		end
		local protected = function(self, mv, camRel)
			if antiBee_originalMoveFunction then return antiBee_originalMoveFunction(self, mv, camRel) end
		end
		Controls.moveFunction = protected
		antiBee_controlsProtected = true
		table.insert(antiBee_connections, game:GetService("RunService").Heartbeat:Connect(function()
			if antiBee_running and Controls.moveFunction ~= protected then
				Controls.moveFunction = protected
			end
		end))
	end)
end

local function antiBee_restoreControls()
	if not antiBee_controlsProtected then return end
	pcall(function()
		local ps = player:FindFirstChild("PlayerScripts")
		local pm = ps and ps:FindFirstChild("PlayerModule")
		if not pm then return end
		local Controls = require(pm):GetControls()
		if Controls and antiBee_originalMoveFunction then
			Controls.moveFunction = antiBee_originalMoveFunction
		end
	end)
	antiBee_controlsProtected = false
end

local function antiBee_blockBuzz()
	pcall(function()
		local ps = player:FindFirstChild("PlayerScripts")
		if not ps then return end
		local bee = ps:FindFirstChild("Bee", true)
		if bee then
			local buzz = bee:FindFirstChild("Buzzing")
			if buzz and buzz:IsA("Sound") then buzz.Volume = 0; buzz:Stop() end
		end
	end)
end

local function antiBee_enable()
	if antiBee_running then return end
	antiBee_running = true
	local L = game:GetService("Lighting")
	for _, inst in ipairs(L:GetDescendants()) do antiBee_nuke(inst) end
	table.insert(antiBee_connections, L.DescendantAdded:Connect(function(obj)
		if antiBee_running then antiBee_nuke(obj) end
	end))
	antiBee_protectControls()
	table.insert(antiBee_connections, game:GetService("RunService").Heartbeat:Connect(function()
		if antiBee_running then antiBee_blockBuzz() end
	end))
	antiBee_startFOV()
end

local function antiBee_disable()
	if not antiBee_running then return end
	antiBee_running = false
	antiBee_restoreControls()
	antiBee_disconnectAll()
	antiBee_stopFOV()
end

		antiBee_enable()

local function espBr_scanPlot(plot)
	if not plot or not plot.Parent then return end
	if espBr_isMyPlot(plot) then
		for uid in pairs(espBrainrot_instances) do
			if uid:find("^"..plot.Name.."_") then espBr_clearESP(uid) end
		end
		return
	end
	local ok, channel = pcall(function() return espBr_Sync:Get(plot.Name) end)
	if not ok or not channel then return end
	local ok2, animalList = pcall(function() return channel:Get("AnimalList") end)
	if not ok2 or not animalList then return end
	local seen = {}
	for slot, animalData in pairs(animalList) do
		if type(animalData) == "table" then
			local info = espBr_AnimalsData and espBr_AnimalsData[animalData.Index]
			local displayName = (info and info.DisplayName) or animalData.Index
			if not displayName then continue end
			local uid = plot.Name.."_"..tostring(slot)
			seen[uid] = true
			local adornee, podium = espBr_getPodiumParts(plot, tostring(slot))
			if adornee then espBr_createESP(uid, adornee, podium, displayName) end
		end
	end
	for uid in pairs(espBrainrot_instances) do
		if uid:find("^"..plot.Name.."_") and not seen[uid] then espBr_clearESP(uid) end
	end
end

	local function silenceClone(c)
		if not c then return end
		pcall(function() c.Transparency = 1 end)
		pcall(function() c.CastShadow = false end)
		pcall(function() c.CanTouch = false end)
		pcall(function() c.CanQuery = false end)
	end

	local function doClone()
		local character = LocalPlayer.Character
		if character and character:FindFirstChild("Humanoid") and character.Humanoid.Health > 0 then
			hip = character.Humanoid.HipHeight
			oldRoot = character:FindFirstChild("HumanoidRootPart")
			if not oldRoot or not oldRoot.Parent then return false end
			for _, c in pairs(oldRoot:GetChildren()) do if c:IsA("Attachment") and (c.Name:find("Beam") or c.Name:find("Attach")) then c:Destroy() end end
			for _, c in pairs(oldRoot:GetChildren()) do if c:IsA("Beam") then c:Destroy() end end
			local tmp = Instance.new("Model"); tmp.Parent = game
			character.Parent = tmp
			clone = oldRoot:Clone(); clone.Parent = character
			oldRoot.Parent = Workspace.CurrentCamera
			clone.CFrame = oldRoot.CFrame
			character.PrimaryPart = clone
			character.Parent = Workspace
			for _, v in pairs(character:GetDescendants()) do
				if v:IsA("Weld") or v:IsA("Motor6D") then
					if v.Part0 == oldRoot then v.Part0 = clone end
					if v.Part1 == oldRoot then v.Part1 = clone end
				end
			end
			tmp:Destroy(); silenceClone(clone); return true
		end
		return false
	end

	local function revertClone()
		local character = LocalPlayer.Character
		if not oldRoot or not oldRoot:IsDescendantOf(Workspace) then return end
		if not character or character.Humanoid.Health <= 0 then return end
		local tmp = Instance.new("Model"); tmp.Parent = game
		character.Parent = tmp; oldRoot.Parent = character
		character.PrimaryPart = oldRoot; character.Parent = Workspace
		oldRoot.CanCollide = true
		for _, v in pairs(character:GetDescendants()) do
			if v:IsA("Weld") or v:IsA("Motor6D") then
				if v.Part0 == clone then v.Part0 = oldRoot end
				if v.Part1 == clone then v.Part1 = oldRoot end
			end
		end
		if clone then local p = clone.CFrame; clone:Destroy(); clone = nil; oldRoot.CFrame = p end
		oldRoot = nil
		if character and character.Humanoid then character.Humanoid.HipHeight = hip end
		clearAllGhosts()
	end

local function createToggle(name, order)
    local Row = Instance.new("Frame")
    Row.Name = name .. "Row"
    Row.Size = UDim2.new(1, 0, 0, 52)
    Row.BackgroundTransparency = 1
    Row.LayoutOrder = order
    Row.Parent = ScrollingFrame

    -- Label
    local Label = Instance.new("TextLabel")
    Label.Size = UDim2.new(1, -80, 1, 0)
    Label.Position = UDim2.new(0, 16, 0, 0)
    Label.BackgroundTransparency = 1
    Label.Text = name
    Label.TextColor3 = Color3.fromRGB(255, 255, 255)
    Label.TextSize = 14
    Label.Font = Enum.Font.Gotham
    Label.TextXAlignment = Enum.TextXAlignment.Left
    Label.TextTruncate = Enum.TextTruncate.AtEnd
    Label.Parent = Row

    -- Toggle background (pill shape)
    local ToggleBg = Instance.new("Frame")
    ToggleBg.Size = UDim2.new(0, 51, 0, 31)
    ToggleBg.Position = UDim2.new(1, -67, 0.5, -15)
    ToggleBg.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
    ToggleBg.BorderSizePixel = 0
    ToggleBg.Parent = Row

    local BgCorner = Instance.new("UICorner")
    BgCorner.CornerRadius = UDim.new(1, 0)
    BgCorner.Parent = ToggleBg

    -- Toggle knob (circle)
    local Knob = Instance.new("Frame")
    Knob.Size = UDim2.new(0, 25, 0, 25)
    Knob.Position = UDim2.new(0, 3, 0.5, -12)
    Knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    Knob.BorderSizePixel = 0
    Knob.Parent = ToggleBg

    local KnobCorner = Instance.new("UICorner")
    KnobCorner.CornerRadius = UDim.new(1, 0)
    KnobCorner.Parent = Knob

    -- Knob shadow effect
    local KnobShadow = Instance.new("UIStroke")
    KnobShadow.Color = Color3.fromRGB(0, 0, 0)
    KnobShadow.Transparency = 0.7
    KnobShadow.Thickness = 1
    KnobShadow.Parent = Knob

    -- Separator line
    local Sep = Instance.new("Frame")
    Sep.Size = UDim2.new(1, -16, 0, 1)
    Sep.Position = UDim2.new(0, 16, 1, -1)
    Sep.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    Sep.BorderSizePixel = 0
    Sep.Parent = Row

    toggleStates[name] = false

    -- Clickable button overlay
    local Button = Instance.new("TextButton")
    Button.Size = UDim2.new(1, 0, 1, 0)
    Button.BackgroundTransparency = 1
    Button.Text = ""
    Button.Parent = Row

    local tweenService = game:GetService("TweenService")
    local tweenInfo = TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

    -- Restore saved visual state immediately (callback fired later by loadSavedStates)
    if type(getgenv().CHLIP_SAVED_STATES) == "table" and getgenv().CHLIP_SAVED_STATES[name] == true then
        toggleStates[name] = true
        Knob.Position = UDim2.new(0, 23, 0.5, -12)
        ToggleBg.BackgroundColor3 = Color3.fromRGB(76, 175, 80)
    end

    Button.MouseButton1Click:Connect(function()
        toggleStates[name] = not toggleStates[name]
        local on = toggleStates[name]

        -- Save the new state immediately
        if type(getgenv().CHLIP_SAVED_STATES) == "table" then
            if not (name == "Desync (Work in Progress)") then
                getgenv().CHLIP_SAVED_STATES[name] = on
                pcall(function()
                    local toSave = {}
                    for k, v in pairs(getgenv().CHLIP_SAVED_STATES) do
                        if k ~= "Desync (Work in Progress)" then toSave[k] = v end
                    end
                    writefile("ChlipHubSave.json", game:GetService("HttpService"):JSONEncode(toSave))
                end)
            end
        end

        -- Animate knob
        tweenService:Create(Knob, tweenInfo, {
            Position = on and UDim2.new(0, 23, 0.5, -12) or UDim2.new(0, 3, 0.5, -12)
        }):Play()

        -- Animate background color
        tweenService:Create(ToggleBg, tweenInfo, {
            BackgroundColor3 = on and Color3.fromRGB(76, 175, 80) or Color3.fromRGB(60, 60, 60)
        }):Play()

        -- Fire feature callback if one exists
        if toggleCallbacks[name] then
            toggleCallbacks[name](on)
        end
    end)

    -- Hover effect
    Button.MouseEnter:Connect(function()
        tweenService:Create(Row, TweenInfo.new(0.1), {
            BackgroundTransparency = 0.95
        }):Play()
        Row.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    end)
    Button.MouseLeave:Connect(function()
        tweenService:Create(Row, TweenInfo.new(0.1), {
            BackgroundTransparency = 1
        }):Play()
    end)
end
